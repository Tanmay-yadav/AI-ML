# OOP Cheat Sheet\n\n## Classes and Objects\n```python\nclass Dog:\n    # Class variable\n    species = \"Canis familiaris\"\n    \n    # Constructor\n    def __init__(self, name, age):\n        # Instance variables\n        self.name = name\n        self.age = age\n    \n    # Instance method\n    def bark(self):\n        return f\"{self.name} says woof!\"\n    \n    # Class method\n    @classmethod\n    def get_species(cls):\n        return cls.species\n    \n    # Static method\n    @staticmethod\n    def general_info():\n        return \"Dogs are great pets\"\n\n# Create instance\ndog = Dog(\"Buddy\", 3)\nprint(dog.bark())\n```\n\n## Inheritance\n```python\nclass Animal:\n    def __init__(self, name):\n        self.name = name\n    \n    def speak(self):\n        return f\"{self.name} makes a sound\"\n\nclass Dog(Animal):\n    def speak(self):\n        return f\"{self.name} barks\"\n    \n    def fetch(self):\n        return f\"{self.name} fetches the ball\"\n\ndog = Dog(\"Buddy\")\nprint(dog.speak())  # Overridden method\nprint(dog.fetch())  # New method\n```\n\n## Special Methods (Dunder Methods)\n```python\nclass MyClass:\n    def __init__(self):\n        # Constructor\n        pass\n    \n    def __str__(self):\n        # String representation (user-friendly)\n        return \"MyClass instance\"\n    \n    def __repr__(self):\n        # Official string representation (developer-friendly)\n        return \"MyClass()\"\n    \n    def __len__(self):\n        # len(obj)\n        return 0\n    \n    def __getitem__(self, key):\n        # obj[key]\n        return None\n    \n    def __setitem__(self, key, value):\n        # obj[key] = value\n        pass\n    \n    def __contains__(self, item):\n        # item in obj\n        return False\n    \n    def __call__(self):\n        # obj()\n        pass\n    \n    def __eq__(self, other):\n        # obj == other\n        return True\n    \n    def __lt__(self, other):\n        # obj < other\n        return False\n```\n\n## Properties\n```python\nclass Circle:\n    def __init__(self, radius):\n        self._radius = radius\n    \n    @property\n    def radius(self):\n        return self._radius\n    \n    @radius.setter\n    def radius(self, value):\n        if value < 0:\n            raise ValueError(\"Radius must be positive\")\n        self._radius = value\n    \n    @radius.deleter\n    def radius(self):\n        del self._radius\n\ncircle = Circle(5)\nprint(circle.radius)  # Get\ncircle.radius = 10    # Set\ndel circle.radius     # Delete\n```\n\n## Encapsulation\n```python\nclass Person:\n    def __init__(self, name, age):\n        self.name = name          # Public\n        self._age = age           # Protected (convention)\n        self.__secret = \"hidden\"  # Private (name mangling)\n    \n    def get_age(self):\n        return self._age\n    \n    def set_age(self, age):\n        if age > 0:\n            self._age = age\n\nperson = Person(\"Alice\", 25)\nprint(person.get_age())\nperson.set_age(26)\n```\n\n## Polymorphism\n```python\nclass Shape:\n    def area(self):\n        raise NotImplementedError\n\nclass Circle(Shape):\n    def __init__(self, radius):\n        self.radius = radius\n    \n    def area(self):\n        return 3.14 * self.radius ** 2\n\nclass Rectangle(Shape):\n    def __init__(self, width, height):\n        self.width = width\n        self.height = height\n    \n    def area(self):\n        return self.width * self.height\n\n# Works with any Shape\nshapes = [Circle(5), Rectangle(3, 4)]\nfor shape in shapes:\n    print(shape.area())\n```\n\n## Abstract Base Classes\n```python\nfrom abc import ABC, abstractmethod\n\nclass Animal(ABC):\n    @abstractmethod\n    def speak(self):\n        pass\n\nclass Dog(Animal):\n    def speak(self):\n        return \"Woof!\"\n\n# Can't instantiate ABC\n# animal = Animal()  # Error\n\ndog = Dog()\nprint(dog.speak())\n```\n\n## Multiple Inheritance\n```python\nclass Flyable:\n    def fly(self):\n        return \"Flying\"\n\nclass Swimmable:\n    def swim(self):\n        return \"Swimming\"\n\nclass Duck(Flyable, Swimmable):\n    pass\n\nduck = Duck()\nprint(duck.fly())\nprint(duck.swim())\n```\n\n## Composition\n```python\nclass Engine:\n    def start(self):\n        return \"Engine started\"\n\nclass Car:\n    def __init__(self):\n        self.engine = Engine()\n    \n    def start(self):\n        return self.engine.start()\n\ncar = Car()\nprint(car.start())\n```\n\n## Dataclasses\n```python\nfrom dataclasses import dataclass\n\n@dataclass\nclass Person:\n    name: str\n    age: int\n    email: str = \"example@example.com\"\n\nperson = Person(\"Alice\", 25)\nprint(person)\nprint(person.name)\n```\n"
